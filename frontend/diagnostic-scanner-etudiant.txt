function getRole(): string {
  try {
    const token = TokenStorage.get();
    if (!token) return '';
    const payload = token.split('.')[1];
    const normalized = payload.replace(/-/g, '+').replace(/_/g, '/');
    const decoded = JSON.parse(atob(normalized));
    return decoded.role ?? decoded.roles?.[0] ?? '';
  } catch {
    return '';
  }
}

function DashboardForRole() {
  return getRole() === 'ETUDIANT'
    ? <Navigate to="/etudiant/annonces" replace />
    : <DashboardPage />;
}

function displayValue(value: unknown): string {
  if (value === null || value === undefined || value === '') return '—';

function formatDate(value: unknown): string {
  if (typeof value !== 'string' || !value) return '—';

  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return value;

  return date.toLocaleDateString('fr-FR');
}

function asArray(result: ApiResult): RecordItem[] {
  if (result.status !== 'fulfilled' || !Array.isArray(result.value.data)) {
    return [];
  }

  return result.value.data.filter(
    (item): item is RecordItem => typeof item === 'object' && item !== null,
  );
}

function extractToken(rawValue: string): string | null {
  const raw = rawValue.trim();
  try {
    const url = new URL(raw);
    const match = url.pathname.match(/\/pointage\/([^/]+)/);
    if (match?.[1]) return decodeURIComponent(match[1]);
  } catch {
    // Le QR peut aussi contenir directement le token.
  }
  if (/^[a-zA-Z0-9_-]{12,}$/.test(raw)) return raw;
  return null;
}

    const token = extractToken(rawValue);
    if (!token) {
      setError('Ce QR code ne correspond pas à une session de pointage COSTM.');
      return;
    }

    setBusy(true);
    setError('');
    setMessage('');
    stopCamera();

    try {
      await scanPresenceSessionUseCase.execute(token);
      setMessage('Présence enregistrée avec succès. Votre pointage est transmis à COSTM.');
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Pointage impossible. Le QR code a peut-être expiré ou a déjà été utilisé.');
    } finally {
      setBusy(false);
    }
  }

  async function startCamera() {
    setError('');
    setMessage('');

    if (!navigator.mediaDevices?.getUserMedia) {
      setError('La caméra nécessite HTTPS ou localhost et une autorisation du navigateur.');
      return;
    }

    if (!videoRef.current) {
      setError('La zone de caméra est introuvable. Rechargez la page et réessayez.');
      return;
    }

    try {
      stopCamera();

      const reader = new BrowserQRCodeReader();
      readerRef.current = reader;
      runningRef.current = true;
      setScanning(true);

      const controls = await reader.decodeFromVideoDevice(
        undefined,
        videoRef.current,
        (result, _scanError, scanControls) => {
          if (result && runningRef.current) {
            runningRef.current = false;
            scanControls.stop();
            void submitScannedValue(result.getText());
          }
        }
      );

      controlsRef.current = controls;

      if (!runningRef.current) {
        controls.stop();
      }
    } catch (e) {
      stopCamera();
      setError(
        e instanceof Error
          ? e.message
          : 'Impossible de démarrer la caméra. Vérifiez les autorisations du navigateur.'
      );
    }
  }

  useEffect(() => () => {
    runningRef.current = false;
    controlsRef.current?.stop();
    streamRef.current?.getTracks().forEach((track) => track.stop());
    if (frameRef.current !== null) cancelAnimationFrame(frameRef.current);
  }, []);

  return (
    <section className="portal-page">
      <PageTitle eyebrow="POINTAGE · COSTM" title="Présence" description="Scannez le QR code présenté par l’administration pour enregistrer votre présence au cours." />
      <section className="portal-scan-card">
        <div className="portal-scan-icon">▦</div>
        <h2>Scanner le QR code du cours</h2>
        <p>Le QR doit être actif. Les sessions de pointage expirent après cinq minutes.</p>
        <video ref={videoRef} className={scanning ? 'portal-camera portal-camera-active' : 'portal-camera'} playsInline muted />
        {!scanning && !busy && (
          <button className="portal-primary-button" type="button" onClick={() => void startCamera()}>
            Ouvrir la caméra
          </button>
        )}
        {scanning && <button className="portal-secondary-button" type="button" onClick={stopCamera}>Arrêter la caméra</button>}
        {busy && <p>Enregistrement de votre présence…</p>}
        {message && <Notice>{message}</Notice>}
        {error && <Notice error>{error}</Notice>}
        <p className="portal-muted">Autorisez l’accès à la caméra lorsque le navigateur le demande.</p>
      </section>
    </section>
  );
}

import { useEffect, useState } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import { scanPresenceSessionUseCase } from '../di/containers';

export default function PointageQrStudentPage() {
  const { token } = useParams<{ token: string }>();
  const navigate = useNavigate();
  const [status, setStatus] = useState('Vérification de la session...');
  const [success, setSuccess] = useState(false);

  useEffect(() => {
    let cancelled = false;

    async function scan() {
      if (!token) {
        setStatus('Le lien du QR code est invalide.');
        return;
      }

      const accessToken = localStorage.getItem('costm_access_token');

      if (!accessToken) {
        navigate('/login', {
          replace: true,
          state: { from: `/pointage/${token}` },
        });
        return;
      }

      try {
        await scanPresenceSessionUseCase.execute(token);
        if (!cancelled) {
          setSuccess(true);
          setStatus('Présence enregistrée avec succès.');
        }
      } catch (e) {
        if (!cancelled) {
          setSuccess(false);
          setStatus(e instanceof Error ? e.message : 'Le pointage a échoué. Le QR code a peut-être expiré ou a déjà été utilisé.');
        }
      }
    }

    void scan();

    return () => {
      cancelled = true;
    };
  }, [token, navigate]);

  return (
    <main style={{ minHeight: '100vh', display: 'grid', placeItems: 'center', padding: 24, background: '#f1f5f9', color: '#172033' }}>
      <section style={{ width: '100%', maxWidth: 480, padding: 28, background: 'white', borderRadius: 16, boxShadow: '0 10px 30px rgba(15,23,42,0.08)', textAlign: 'center' }}>
        <h1 style={{ marginTop: 0 }}>Pointage COSTM</h1>
        <p style={{ fontSize: 17, lineHeight: 1.6 }}>{status}</p>
        {success && <p style={{ color: '#15803d', fontWeight: 700 }}>Tu peux maintenant retourner à tes activités.</p>}
        {!success && !status.includes('Vérification') && (
          <button type="button" onClick={() => navigate('/login')} style={{ padding: '10px 16px', border: 0, borderRadius: 8, background: '#1d4ed8', color: 'white', cursor: 'pointer' }}>
            Retour à la connexion
          </button>
        )}
      </section>
    </main>
  );
}

import { apiClient } from '../http/apiClient';
import type { PresenceSession } from '../../domain/entities/PresenceSession';
import type { PresenceSessionRepository } from '../../domain/repositories/PresenceSessionRepository';

export class HttpPresenceSessionRepository implements PresenceSessionRepository {
  async create(cours: string): Promise<PresenceSession> {
    const response = await apiClient.post<PresenceSession>(
      '/presences/sessions',
      { cours },
    );
    return response.data;
  }

  async list(): Promise<PresenceSession[]> {
    const response = await apiClient.get<PresenceSession[]>('/presences/sessions');
    return response.data;
  }

  async close(id: string): Promise<PresenceSession> {
    const response = await apiClient.patch<PresenceSession>(
      `/presences/sessions/${id}/close`,
    );
    return response.data;
  }

  async scan(token: string): Promise<unknown> {
    const response = await apiClient.post(
      `/presences/sessions/${encodeURIComponent(token)}/scan`,
    );
    return response.data;
  }
}
